using Microsoft.EntityFrameworkCore;
using MovieAPI.Models;

public class MovieAPIContextOld : DbContext
{
    public MovieAPIContextOld(DbContextOptions<MovieAPIContextOld> options) : base(options)
    {
    }

    public DbSet<Movie> Movie { get; set; } = default!;
    public DbSet<Genre> Genre { get; set; } = default!;
    public DbSet<MovieDetails> MovieDetails { get; set; } = default!;
    public DbSet<Review> Review { get; set; } = default!;
    public DbSet<Actor> Actor { get; set; } = default!;
    public DbSet<Temp_MovieActor> MovieActor { get; set; } = default!;
}
